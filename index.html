export async function matchInsuranceAgents(userQuery: string) {
  const queryVector = await generateEmbedding(userQuery);
  
  const { data: agents, error } = await supabase
    .rpc('match_agent_profiles', {

    });

class YOLOv8Detector(private val context: Context) {
  private var interpreter: Interpreter? = null

  fun detectFrame(bitmap: Bitmap): List<Detection> {
    val inputTensor = ImageProcessor.process(bitmap)
    val outputLocations = Array(1) { Array(84) { FloatArray(8400) } }
    
    interpreter?.run(inputTensor.buffer, outputLocations)
    return nonMaxSuppression(outputLocations, threshold = 0.65f)
  }
}
